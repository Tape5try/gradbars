\usepackage{gradbars}
\usegradbarslibrary{reports,distribution}
\setlength{\parindent}{0pt}
\ExplSyntaxOn
\gradbarsdeclarebackend{correlation-check}{bar,heat,missing,signed,gradient,print,labels,units,clip,decorations,embedded,tagging,artifact}{

\str_if_eq:eeF{\gradbarsbackendvalue{heatpalette}}{diverging}{\errmessage{Correlation~palette~changed}}
\gradbarsbackenddefault}
\ExplSyntaxOff
