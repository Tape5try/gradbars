\documentclass{article}
\usepackage{gradbars}
\makeatletter
% Audit known endpoint positions in a rendered picture, independent of the
% arithmetic used to obtain them. Expected positions are fractions of width.
\newcommand\checkbar[6]{%
  \begingroup
  \let\originaldraw\gradbars@draw
  \def\gradbars@draw{%
    \pgfmathparse{abs(\gradbars@zero-(#3))<0.00001}%
    \ifdim\pgfmathresult pt<.5pt \errmessage{Wrong zero position}\fi
    \pgfmathparse{abs(\gradbars@end-(#4))<0.00001}%
    \ifdim\pgfmathresult pt<.5pt \errmessage{Wrong value endpoint}\fi
    \edef\actualcolor{\gradbars@left}\edef\expectedcolor{#5}%
    \ifx\actualcolor\expectedcolor\else\errmessage{Wrong conditional color}\fi
    \originaldraw
    #6%
  }%
  \gradbar[#1]{#2}\par
  \endgroup
}
\begin{document}
\checkbar{min=-100,max=100}{-50}{.5}{.25}{lightred}{}
\checkbar{min=-100,max=100}{50}{.5}{.75}{gradbarsBlue}{}
\checkbar{min=-20,max=80}{-10}{.2}{.1}{lightred}{}
\checkbar{thresholds={60,80},threshold colors={red,green,blue}}{59}{0}{.59}{red}{}
\checkbar{thresholds={60,80},threshold colors={red,green,blue}}{60}{0}{.6}{green}{}
\checkbar{thresholds={60,80},threshold colors={red,green,blue}}{80}{0}{.8}{blue}{}
\checkbar{min=-50,max=50,error minus=8,error plus=12}{-5}{.5}{.45}{lightred}{%
  \pgfmathparse{abs(\gradbars@errorlo-.37)<.00001}%
  \ifdim\pgfmathresult pt<.5pt \errmessage{Wrong lower error endpoint}\fi
  \pgfmathparse{abs(\gradbars@errorhi-.57)<.00001}%
  \ifdim\pgfmathresult pt<.5pt \errmessage{Wrong upper error endpoint}\fi
}
\checkbar{width=100pt,rounded=10pt}{1}{0}{.01}{gradbarsBlue}{%
  \ifdim\gradbars@fillradius>.501pt \errmessage{Tiny bar radius is too large}\fi
  \ifdim\gradbars@xstop>1.001pt \errmessage{Tiny bar was inflated}\fi
}
\checkbar{target=80}{72}{0}{.72}{gradbarsBlue}{%
  \pgfmathparse{abs(\gradbars@targetpos-.8)<.00001}%
  \ifdim\pgfmathresult pt<.5pt \errmessage{Wrong target position}\fi
}
% The options used by checkbar must not leak into later calls.
\checkbar{}{50}{0}{.5}{gradbarsBlue}{}
\end{document}
