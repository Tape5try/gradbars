\usepackage{gradbars}
\pagestyle{empty}
\newsavebox{\zerobar}
\newsavebox{\fullbar}
\newsavebox{\smallbar}
\newsavebox{\localbar}
\begin{document}
% Labels are independently checked with pdftotext by tests/run.py.

PERCENT: \gradbar[max=200,value format=percent,precision=1]{50}\par
UNIT: \gradbar[max=200,precision=1,unit={\,ms}]{50}\par
LARGE: \gradbar[max=125000,precision=0]{87500}\par
CUSTOM: \gradbar[max=10,text={eight of ten}]{8}\par

\sbox{\fullbar}{\gradbar{100}}
\ifdim\wd\zerobar=\wd\fullbar\else
  \errmessage{Bar widths vary with their values}
\fi
% Local overrides and a grouped setup must leave subsequent bars unchanged.
\sbox{\localbar}{\gradbar[width=40mm,label width=7em]{40}}
\begingroup\gradbarssetup{width=50mm}\sbox{\localbar}{\gradbar{10}}\endgroup
\sbox{\smallbar}{\gradbar{10}}
\ifdim\wd\smallbar=\wd\zerobar\else
  \errmessage{Local configuration leaked}
\fi
% Labels disabled: no node padding may enlarge the requested track width.
\sbox{\smallbar}{\gradbar[width=20mm,label=none]{100}}
\ifdim\wd\smallbar>20.01mm \errmessage{Bar width exceeds requested length}\fi
\ifdim\wd\smallbar<19.99mm \errmessage{Bar width is shorter than requested length}\fi
\end{document}
